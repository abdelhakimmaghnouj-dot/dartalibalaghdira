        // --- ÉTAT DU PANIER ---
        let cart = [];

        // --- FONCTIONS D'AFFICHAGE ---

        // Initialisation
        document.addEventListener('DOMContentLoaded', () => {
            renderProducts();
            loadCart();
        });

        function renderProducts() {
            const grid = document.getElementById('productGrid');
            grid.innerHTML = products.map(product => `
                <div class="product-card">
                    <img src="${product.image}" alt="${product.name}" class="product-image" loading="lazy">
                    <div class="product-info">
                        <h3 class="product-title">${product.name}</h3>
                        <p class="product-desc">${product.description}</p>
                        <div class="product-footer">
                            <span class="price">${formatPrice(product.price)}</span>
                            <button class="btn btn-primary" onclick="addToCart(${product.id})">
                                Ajouter
                            </button>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // --- GESTION DU PANIER ---

        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            const existingItem = cart.find(item => item.id === productId);

            if (existingItem) {
                existingItem.qty++;
            } else {
                cart.push({ ...product, qty: 1 });
            }

            saveCart();
            updateCartUI();
            showToast(`"${product.name}" ajouté au panier`);
            
            // Ouvrir le panier automatiquement au premier ajout pour feedback immédiat
            if (cart.length === 1 && cart[0].qty === 1) {
                // Optionnel : openCart(); 
            }
        }

        function removeFromCart(productId) {
            cart = cart.filter(item => item.id !== productId);
            saveCart();
            updateCartUI();
        }

        function updateQty(productId, change) {
            const item = cart.find(i => i.id === productId);
            if (item) {
                item.qty += change;
                if (item.qty <= 0) {
                    removeFromCart(productId);
                } else {
                    saveCart();
                    updateCartUI();
                }
            }
        }

        function updateCartUI() {
            // Mise à jour du compteur
            const totalQty = cart.reduce((acc, item) => acc + item.qty, 0);
            document.getElementById('cartCount').textContent = totalQty;

            // Mise à jour de la liste
            const cartItemsContainer = document.getElementById('cartItems');
            const checkoutSection = document.getElementById('checkoutSection');

            if (cart.length === 0) {
                cartItemsContainer.innerHTML = '<p style="text-align: center; color: var(--text-light); margin-top: 2rem;">Votre panier est vide.</p>';
                checkoutSection.style.display = 'none';
                return;
            }

            checkoutSection.style.display = 'block';
            cartItemsContainer.innerHTML = cart.map(item => `
                <div class="cart-item">
                    <div class="item-details">
                        <h4>${item.name}</h4>
                        <span>${formatPrice(item.price)} x ${item.qty} = <strong>${formatPrice(item.price * item.qty)}</strong></span>
                    </div>
                    <div class="item-controls">
                        <button class="qty-btn" onclick="updateQty(${item.id}, -1)">-</button>
                        <span>${item.qty}</span>
                        <button class="qty-btn" onclick="updateQty(${item.id}, 1)">+</button>
                        <button class="remove-btn" onclick="removeFromCart(${item.id})" title="Supprimer">🗑️</button>
                    </div>
                </div>
            `).join('');

            // Calcul du total
            const total = cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
            document.getElementById('cartTotal').textContent = formatPrice(total);
        }

        // --- GESTION DU FORMULAIRE ET ENVOI ---

        function submitOrder(e) {
            e.preventDefault();

            // Récupération des données
            const lastname = document.getElementById('lastname').value.trim();
            const firstname = document.getElementById('firstname').value.trim();
            const phone = document.getElementById('phone').value.trim();

            if (!lastname || !firstname || !phone) {
                showToast("Veuillez remplir tous les champs.");
                return;
            }

            // Construction du message pour WhatsApp
            let message = `*NOUVELLE COMMANDE* \n\n`;
            message += `*Client :* ${firstname} ${lastname}\n`;
            message += `*Téléphone :* ${phone}\n\n`;
            message += `*Détail de la commande :*\n`;

            let total = 0;
            cart.forEach(item => {
                const subtotal = item.price * item.qty;
                total += subtotal;
                message += `- ${item.name} (x${item.qty}): ${formatPrice(subtotal)}\n`;
            });

            message += `\n*TOTAL À PAYER :* ${formatPrice(total)}`;
            message += `\n\n_Merci de me rappeler pour confirmer la commande._`;

            // Encodage pour URL
            const encodedMessage = encodeURIComponent(message);
            const whatsappUrl = `https://wa.me/${BUSINESS_PHONE}?text=${encodedMessage}`;

            // Simulation d'ouverture
            window.open(whatsappUrl, '_blank');
            
            // Optionnel : Vider le panier après commande
            // cart = [];
            // saveCart();
            // updateCartUI();
            // closeCart();
        }

        // --- UTILITAIRES ---

        function formatPrice(price) {
            return price.toFixed(2).replace('.', ',') + ' ' + CURRENCY;
        }

        function openCart() {
            document.getElementById('cartModal').classList.add('active');
        }

        function closeCart() {
            document.getElementById('cartModal').classList.remove('active');
        }

        // Fermer le modal si on clique dehors
        document.getElementById('cartModal').addEventListener('click', (e) => {
            if (e.target === document.getElementById('cartModal')) {
                closeCart();
            }
        });

        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }

        // Persistance simple (LocalStorage)
        function saveCart() {
            localStorage.setItem('myShopCart', JSON.stringify(cart));
        }

        function loadCart() {
            const savedCart = localStorage.getItem('myShopCart');
            if (savedCart) {
                cart = JSON.parse(savedCart);
                updateCartUI();
            }
        }
